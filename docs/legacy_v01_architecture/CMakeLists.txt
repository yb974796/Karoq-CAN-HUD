cmake_minimum_required(VERSION 3.16)
project(KaroqHudPortableCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(hud_core core/can_profiles.cpp)
target_include_directories(hud_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/core")
add_executable(hud_demo tests/demo_main.cpp)
target_link_libraries(hud_demo PRIVATE hud_core)
add_executable(test_decoder tests/test_decoder.cpp)
target_link_libraries(test_decoder PRIVATE hud_core)
enable_testing()
add_test(NAME portable_core_test COMMAND test_decoder "${CMAKE_CURRENT_SOURCE_DIR}/profiles/skoda_karoq_2019/can_extended_reference.csv")
